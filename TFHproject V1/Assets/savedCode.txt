
		function generateCubes(number: Number): void {
			
			addCube(500,0,0,0);
			var i: Number = 2;
			while( i <= number ) {
				var x: Number = Math.ceil( Math.random() * 12000) - (root.width / 2);
				var z: Number = Math.ceil( Math.random() * 12000);
				addCube(500,x,0,z);
				i++;
			}
		}
		function addCube(size,x,y,z :Number): void {
			
			var textur       :BitmapAssetMaterial = new BitmapAssetMaterial( "Cubetextur" );
			var materialList :MaterialsList       = new MaterialsList(null);
		    materialList.addMaterial( textur, "front" );
			materialList.addMaterial( textur, "back" );
			materialList.addMaterial( textur, "right" );
			materialList.addMaterial( textur, "left" );
			materialList.addMaterial( textur, "top" );
			materialList.addMaterial( textur, "bottom" );

			var kubus = new Cube( materialList, size, size, size, 16, 16, 16, 0.5 );
			kubus.x = x;
			kubus.y = y;
			kubus.z = z;
			scene.addChild( kubus );
			
			
		}
